# In euer bestehendes Caddyfile aufnehmen, dann Caddy neu laden:
#   docker exec <caddy-container> caddy reload --config /etc/caddy/Caddyfile
# Voraussetzung: DNS-A-Records n8n, hub, media → Server-IP; Container im selben Netz wie Caddy.

# ---------------------------------------------------------------- n8n (Stufe 1)
n8n.nettverwaltet.de {
    encode gzip

    # Webhooks, Formulare und Wartepunkte müssen von außen erreichbar sein
    # (Telegram, Meta, Formular-Eingaben, "Send and wait"-Links).
    @offen path /webhook/* /webhook-test/* /webhook-waiting/* /form/* /form-test/* /form-waiting/* /healthz
    handle @offen {
        reverse_proxy contenthub-n8n:5678
    }

    # Die Oberfläche selbst zusätzlich absichern (eines von beiden wählen):
    #   a) IP-Allowlist – eigene IP(s) eintragen
    #   b) basicauth – Hash erzeugen mit: docker exec <caddy-container> caddy hash-password
    handle {
        # @erlaubt remote_ip 203.0.113.10
        # abort not @erlaubt
        # basicauth {
        #     ritva $2a$14$...
        # }
        reverse_proxy contenthub-n8n:5678
    }
}

# ---------------------------------------------------------------- NocoDB (Stufe 2)
hub.nettverwaltet.de {
    encode gzip
    reverse_proxy contenthub-nocodb:8080
}

# ---------------------------------------------------------------- Medien (Stufe 4/5)
# Instagram/Facebook laden Bilder und Videos ausschließlich über öffentliche URLs.
# Nur lesen, kein Verzeichnis-Listing; Dateinamen enthalten einen Zufallsanteil.
media.nettverwaltet.de {
    root * /srv/contenthub-media
    file_server
    header Cache-Control "public, max-age=86400"
    header X-Content-Type-Options nosniff
    @kein_listing path_regexp ^/[^.]*/?$
    respond @kein_listing 404
}
# Dazu im Caddy-Compose ein Volume:  /opt/contenthub/data/media:/srv/contenthub-media:ro
