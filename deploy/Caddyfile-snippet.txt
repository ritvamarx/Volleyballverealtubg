#   docker exec caddy caddy reload --config /etc/caddy/Caddyfile
# TLS holt Caddy automatisch, sobald der DNS-Record auf die Server-IP zeigt.
#
# Wichtig: "volleyball-app" ist der Container-Name – Caddy und die App müssen
# im selben Docker-Netz hängen (Netzname in docker-compose.yml, Standard "caddy").
